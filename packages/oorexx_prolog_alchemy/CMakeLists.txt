cmake_minimum_required(VERSION 3.16)
project(oorexx_prolog_alchemy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_path(OOREXX_INCLUDE_DIR oorexxapi.h HINTS ENV OOREXX_INCLUDE_DIR PATHS /usr/local/include /usr/include)
find_library(OOREXX_LIBRARY rexx HINTS ENV OOREXX_LIB_DIR PATHS /usr/local/lib /usr/lib)
find_path(SWIPL_INCLUDE_DIR SWI-Prolog.h HINTS ENV SWIPL_INCLUDE_DIR PATH_SUFFIXES swi-prolog/include include)
find_library(SWIPL_LIBRARY swipl HINTS ENV SWIPL_LIB_DIR)
if(NOT OOREXX_INCLUDE_DIR OR NOT OOREXX_LIBRARY)
  message(FATAL_ERROR "ooRexx development headers/library not found; set OOREXX_INCLUDE_DIR/OOREXX_LIB_DIR")
endif()
if(NOT SWIPL_INCLUDE_DIR OR NOT SWIPL_LIBRARY)
  message(FATAL_ERROR "SWI-Prolog development headers/library not found; set SWIPL_INCLUDE_DIR/SWIPL_LIB_DIR")
endif()
add_library(prolog_alchemy_native SHARED native/prolog_alchemy_native.cpp)
target_include_directories(prolog_alchemy_native PRIVATE ${OOREXX_INCLUDE_DIR} ${SWIPL_INCLUDE_DIR})
target_link_libraries(prolog_alchemy_native PRIVATE ${OOREXX_LIBRARY} ${SWIPL_LIBRARY})
set_target_properties(prolog_alchemy_native PROPERTIES PREFIX "lib")
# Demo handoff: resolve bundled SWI relative to lib/libprolog_alchemy_native.so.
set_target_properties(prolog_alchemy_native PROPERTIES
  BUILD_RPATH "\$ORIGIN/../swipl/lib/swipl/lib/x86_64-linux"
  INSTALL_RPATH "\$ORIGIN/../swipl/lib/swipl/lib/x86_64-linux")
install(TARGETS prolog_alchemy_native LIBRARY DESTINATION lib)
