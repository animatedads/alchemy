cmake_minimum_required(VERSION 3.21)
project(wire_ui_windows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(OOREXX_ROOT "" CACHE PATH "Portable ooRexx Windows root (contains include/, lib/, bin/)")
if(NOT OOREXX_ROOT)
  message(FATAL_ERROR "Set OOREXX_ROOT to the portable ooRexx Windows distribution")
endif()
add_library(wire_windows_renderer STATIC src/wire_windows_renderer.cpp)
target_include_directories(wire_windows_renderer PUBLIC include)
target_link_libraries(wire_windows_renderer PUBLIC comctl32)
add_executable(wire_ui_windows WIN32 src/wire_windows_host.cpp src/wire_windows_rexx_host.cpp)
target_include_directories(wire_ui_windows PRIVATE include "${OOREXX_ROOT}/include")
target_link_directories(wire_ui_windows PRIVATE "${OOREXX_ROOT}/lib")
target_link_libraries(wire_ui_windows PRIVATE wire_windows_renderer comctl32 rexx rexxapi)
