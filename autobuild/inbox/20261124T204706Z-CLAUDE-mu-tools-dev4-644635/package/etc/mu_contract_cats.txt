# v0.3: the programs on the MU.  A program needs a witness, by name or by its package.
# Names under "forbid exec" are reported even when their package is allowed.
check exec /usr/bin /usr/sbin /usr/local/bin /usr/local/sbin | the PATH of root and of the bring-up account
allow exec-package systemd udev | init until the static init (stage D)
allow exec agetty login | the serial console, break-glass
allow exec-package openssh-server openssh-sftp-server | bring-up channel, until M8 (sshnode.sh runs commands and transfers files)
allow exec-package bash dash | the shells sshnode.sh and systemd units run, until M8 and stage D
allow exec-package sudo | sshnode.sh orders run "sudo", until M8
allow exec-package chrony | time
allow exec-package nftables | the fence (newshell reads it with nft -a list ruleset)
allow exec getent ldd | newshell readings (accounts) and the kit self-test's library check
allow exec-package oorexx | mu-agent, mu_audit and newshell
allow exec-package coreutils tar gzip grep procps iproute2 | the bring-up orders sent through sshnode.sh (sha256sum, cat, ls, tar -z, grep, pgrep, ss), until M8
allow exec dbus-daemon | with systemd-logind, for ssh login sessions only, until M8
allow exec awk | the alternatives link to mawk, which dpkg maintainer scripts use
allow exec-package apparmor | SecOp's MU profile set (apparmor.service is witnessed)
allow exec-package gpgv | apt checks the signatures on package indexes, until the image
allow exec-package debconf ucf sensible-utils adduser debianutils diffutils findutils sed mawk init-system-helpers libc-bin login passwd libpam-modules-bin libpam-runtime base-files base-passwd install-info ncurses-bin hostname bsdutils sysvinit-utils util-linux tzdata | dpkg, its maintainer scripts and the base system, until the image
allow exec-package systemd-sysv | init: PID 1 is /sbin/init, until stage D
allow exec-package mount e2fsprogs logsave kmod | systemd mounts, checks and loads modules at boot, until stage D
allow exec-package grub-common grub2-common grub-pc grub-gfxpayload-lists initramfs-tools initramfs-tools-core linux-base efibootmgr mokutil sbsigntool shim-signed cpio zstd xz-utils dosfstools | the boot chain (GRUB, initramfs, signed kernel, EFI partition), until stage C
allow exec-package azure-vm-utils nvme-cli | Azure names its NVMe disks through these udev rules; until the image
allow exec-package uuid-runtime | nvme-cli depends on it (apt-get -s, REQ-00005739); until the image
allow exec-package libcap2-bin | chrony and iproute2 depend on it (apt-get -s, REQ-00005739)
allow exec-package gettext-base | the signed boot chain (shim-signed, grub) cannot be resolved without it (apt-get -s, REQ-00005739); until stage C
allow exec-package cryptsetup-bin dmsetup | removal rebuilds the initramfs; not worth a boot risk before stage C, when the boot chain is ours
allow exec-package openssh-client | openssh-server depends on it (apt-get -s, REQ-00005739); until M8
allow exec-package dpkg apt | the package manager, until the image (same witness as /var/lib/dpkg in mu_tree_rules.txt)

forbid exec personality setarch i386 x86_64 linux32 linux64 uname26
